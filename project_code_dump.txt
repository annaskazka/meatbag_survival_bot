import asyncio
import logging
import sys

from aiogram import Bot, Dispatcher, html
from aiogram.client.default import DefaultBotProperties
from aiogram.enums import ParseMode
from aiogram.exceptions import TelegramForbiddenError
from aiogram.types import ErrorEvent

from config import BOT_TOKEN
from database.base import Base, engine
from database.requests import init_exercises, set_user_inactive  # <--- Добавили set_user_inactive сюда
from handlers.user_commands import user_router


async def main() -> None:
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

    await init_exercises()

    bot = Bot(token=BOT_TOKEN, default=DefaultBotProperties(parse_mode=ParseMode.HTML))
    dp = Dispatcher()

    # Глобальный обработчик ошибок: гасит блокировки и ставит флаг неактивности в БД
    @dp.error()
    async def global_error_handler(event: ErrorEvent):
        if isinstance(event.exception, TelegramForbiddenError):
            if event.update.message and event.update.message.from_user:
                await set_user_inactive(event.update.message.from_user.id)
            elif event.update.callback_query and event.update.callback_query.from_user:
                await set_user_inactive(event.update.callback_query.from_user.id)

            return True  # Тихо гасим ошибку, бот не падает
        raise event.exception

    dp.include_router(user_router)

    print("Бот запущен и ждет сообщения...")

    await dp.start_polling(bot)


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, stream=sys.stdout)
    asyncio.run(main())

from pathlib import Path

# Список папок и расширений, которые нужно исключить из обхода
IGNORE_DIRS = {'.git', '.idea', 'venv', '.venv', '__pycache__', '.pytest_cache', 'build', 'dist', 'migrations'}
IGNORE_EXTENSIONS = {'.pyc', '.png', '.jpg', '.jpeg', '.gif', '.ico', '.pdf', '.zip', '.sqlite3', '.db'}
OUTPUT_FILE = 'project_code_dump.txt'


def bundle_project(root_dir='.'):
    root_path = Path(root_dir).resolve()
    output_file_path = root_path / OUTPUT_FILE

    files_count = 0

    with open(output_file_path, 'w', encoding='utf-8') as out_f:
        for path in sorted(root_path.rglob('*')):
            # Пропускаем директории и сам файл отчета
            if path.is_dir() or path.resolve() == output_file_path.resolve():
                continue

            # Проверяем, входит ли файл в игнорируемые директории
            relative_path = path.relative_to(root_path)
            if any(part in IGNORE_DIRS for part in relative_path.parts):
                continue

            # Проверяем расширение файла
            if path.suffix.lower() in IGNORE_EXTENSIONS:
                continue

            # Записываем разделитель и путь к файлу
            out_f.write(f"{'=' * 80}\n")
            out_f.write(f"ФАЙЛ: {relative_path}\n")
            out_f.write(f"{'=' * 80}\n\n")

            try:
                with open(path, 'r', encoding='utf-8', errors='ignore') as in_f:
                    out_f.write(in_f.read())
                out_f.write("\n\n")
                files_count += 1
            except Exception as e:
                out_f.write(f"[Не удалось прочитать файл: {e}]\n\n")

    print(f"Готово! Обработано файлов: {files_count}")
    print(f"Результат сохранен в: {output_file_path}")


if __name__ == '__main__':
    bundle_project()

from sqlalchemy import String, BigInteger, Integer, Boolean, ForeignKey, func

from .base import Base


class User(Base):
    __tablename__ = "users"

    id: Mapped[int] = mapped_column(primary_key=True)
    telegram_id: Mapped[int] = mapped_column(BigInteger, unique=True, nullable=False)
    username: Mapped[str | None] = mapped_column(String)
    display_name: Mapped[str | None] = mapped_column(String(50))
    stars: Mapped[int] = mapped_column(Integer, default=0)
    is_active: Mapped[bool] = mapped_column(Boolean, default=True)

    logs: Mapped[list["ActivityLog"]] = relationship(back_populates="user")


class Exercise(Base):
    __tablename__ = "exercises"

    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(100), unique=True, nullable=False)
    stars_reward: Mapped[int] = mapped_column(default=1)
    is_repeatable: Mapped[bool] = mapped_column(Boolean, default=True)

    logs: Mapped[list["ActivityLog"]] = relationship(back_populates="exercise")


class ActivityLog(Base):
    __tablename__ = "workout_logs"

    id: Mapped[int] = mapped_column(primary_key=True)
    user_id: Mapped[int] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"))
    exercise_id: Mapped[int] = mapped_column(ForeignKey("exercises.id", ondelete="CASCADE"))
    created_at: Mapped[datetime] = mapped_column(server_default=func.now())

    user: Mapped["User"] = relationship(back_populates="logs")
    exercise: Mapped["Exercise"] = relationship(back_populates="logs")


class PendingAchievement(Base):
    __tablename__ = "pending_achievements"

    id: Mapped[int] = mapped_column(primary_key=True)
    user_id: Mapped[int] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"))
    text: Mapped[str] = mapped_column(String(200), nullable=False)

    user: Mapped["User"] = relationship()

async def set_user(telegram_id: int, username: str | None = None) -> User:
    async with async_session() as session:
        stmt = select(User). where(User.telegram_id == telegram_id)
        result = await session.execute(stmt)
        user = result.scalar_one_or_none()

        if not user:
            user = User(telegram_id=telegram_id, username=username)
            session.add(user)
            await session.commit()
            await session.refresh(user)

        return user

async def get_all_exercises():
    """Возвращает список всех полезных привычек из базы, отсортированныъх по ценности"""
    async with async_session() as session:
        # Сортируем сначала по количеству звезд, потом по ID
        stmt = select(Exercise).order_by(
            (Exercise.stars_reward == 0).asc(),
            Exercise.stars_reward.asc(),
            Exercise.id.asc())
        result = await session.execute(stmt)
        return result.scalars().all()


async def get_top_users(limit: int = 10):
    """Возвращает топ 10 пользователей по количеству звезд"""
    async with async_session() as session:
        stmt = select(User).order_by(User.stars.desc(), User.id.asc()).limit(limit)
        result = await session.execute(stmt)
        return result.scalars().all()


async def log_activity(telegram_id: int, exercise_id: int):
    """Записывает выполненную активность в лог и начисляет звезды"""

    async with async_session() as session:
        #Находим юзера
        user_stmt = select(User).where(User.telegram_id == telegram_id)
        user_res = await session.execute(user_stmt)
        user = user_res.scalar_one_or_none()

        #Находим упражнение
        ex_stmt = select(Exercise).where(Exercise.id == exercise_id)
        ex_res = await session.execute(ex_stmt)
        exercise = ex_res.scalar_one_or_none()

        if user and exercise:
            #Создаем запись в логе
            log = ActivityLog(user_id=user.id, exercise_id=exercise_id)
            session.add(log)
            #Начисляем звезды, если это не кастомное достижение в 0 звезд
            if exercise.stars_reward > 0:
                user.stars += exercise.stars_reward

            await session.commit()
            return user.stars, exercise
        return 0, None


async def create_pending_achievement(telegram_id: int, text: str) -> int:
    """Сохраняет кастомное достижение на модерацию и возвращает ID заявки"""
    async with async_session() as session:
        user_stmt = select(User).where(User.telegram_id == telegram_id)
        user_res = await session.execute(user_stmt)
        user = user_res.scalar_one_or_none()

        if user:
            pending = PendingAchievement(user_id=user.id, text=text)
            session.add(pending)
            await session.commit()
            await session.refresh(pending)
            return pending.id
        return 0


async def approve_achievement(pending_id: int, reward_stars: int = 3, custom_text: str | None = None):
    """Одобряет заявкуЖ создает новое упражнение для всех и начисляет звезды автору"""
    async with async_session() as session:
        stmt = select(PendingAchievement).where(PendingAchievement.id == pending_id)
        res = await session.execute(stmt)
        pending = res.scalar_one_or_none()

        if not pending:
            return None, None

        # Подтягиваем пользователя по foreign key
        user_stmt = select(User).where(User.id == pending.user_id)
        user_res = await session.execute(user_stmt)
        user = user_res.scalar_one_or_none()

        # Если админ указал новый текст, берем его, иначе текст из заявки
        exercise_name = custom_text.strip() if custom_text else pending.text

        new_exercise = Exercise(name=exercise_name, stars_reward=reward_stars)
        session.add(new_exercise)

        await session.flush()

        #Начисляем звезды автору заявки
        if user:
            user.stars += reward_stars

            #Также запишем в лог активности
            session.add(ActivityLog(user_id=user.id, exercise_id=new_exercise.id))

        #Удаляем заявку из таблицы ожидания
        await session.delete(pending)
        await session.commit()

        return user.telegram_id if user else None, exercise_name


async def reject_achievement(pending_id: int):
    """Отклоняет заявку без наград"""
    async with async_session() as session:
        stmt = select(PendingAchievement).where(PendingAchievement.id == pending_id)
        res = await session.execute(stmt)
        pending = res.scalar_one_or_none()

        if pending:
            #Узнаем telegram_id автора перед удалением
            user_stmt = select(User).where(User.id == pending.user_id)
            user_res = await session.execute(user_stmt)
            user = user_res.scalar_one_or_none()
            telegram_id = user.telegram_id if user else None

            await session.delete(pending)
            await session.commit()
            return telegram_id
        return None

@user_router.message(AdminApproval.waiting_for_stars)
async def process_admin_stars_input(message: Message, state: FSMContext):
    if message.from_user.id != ADMIN_TELEGRAM_ID:
        return

    text = message.text.strip()
    custom_text = None

    if "|" in text:
        parts = text.split("|", 1)
        custom_text = parts[0].strip()
        stars_str = parts[1].strip()
    else:
        stars_str = text

    if not stars_str.isdigit():
        await message.answer(
            "⚠️ Неверный формат!\n"
            "Отправь просто число (например: <code>3</code>)\n"
            "или новый текст с количеством через черту (например: <code>Пройти 10к шагов | 4</code>)."
        )
        return

    reward_stars = int(stars_str)
    data = await state.get_data()
    pending_id = data.get("pending_id")

    await state.clear()

    # Запускаем одобрение в базе с введенным количеством звезд
    author_telegram_id, achievement_text = await approve_achievement(
        pending_id,
        reward_stars=reward_stars,
        custom_text=custom_text
    )

    if author_telegram_id:
        try:
            await message.bot.send_message(
                chat_id=author_telegram_id,
                text=f"🎉 <b>Твое достижение одобрено!</b>\n\n"
                     f"<i>«{achievement_text}»</i>\n"
                     f"Оно добавлено в общий список привычек для всех, а тебе начислено <b>+{reward_stars}⭐</b>!"
            )
        except Exception:
            pass

    await message.answer(
        f"✅ <b>Успешно добавлено в общий список!</b>\n"
        f"Достижение: <i>«{achievement_text}»</i>\n"
        f"Награда: <b>+{reward_stars}⭐</b>"
    )

def get_exercise_kb(exercises) -> InlineKeyboardMarkup:
    builder = InlineKeyboardBuilder()
    for ex in exercises:
        builder.row(
            InlineKeyboardButton(
                text=f"{ex.name} +{ex.stars_reward}⭐" if ex.stars_reward > 0 else ex.name, callback_data=f"ex_{ex.id}"
            )
        )
    return builder.as_markup()
