import asyncio
import logging
import sys

from aiogram import Bot, Dispatcher, html
from aiogram.client.default import DefaultBotProperties
from aiogram.enums import ParseMode

from config import BOT_TOKEN
from database.base import Base, engine
from database.requests import init_exercises
from handlers.user_commands import user_router


async def main() -> None:
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

    await init_exercises()

    bot = Bot(token=BOT_TOKEN, default=DefaultBotProperties(parse_mode=ParseMode.HTML))
    dp = Dispatcher()

    dp.include_router(user_router)

    print("Бот запущен и ждет сообщения...")

    await dp.start_polling(bot)


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, stream=sys.stdout)
    asyncio.run(main())

from datetime import datetime
from sqlalchemy.orm import Mapped, mapped_column, relationship
from sqlalchemy import String, BigInteger, Integer, ForeignKey, func

from .base import Base


class User(Base):
    __tablename__ = "users"

    id: Mapped[int] = mapped_column(primary_key=True)
    telegram_id: Mapped[int] = mapped_column(BigInteger, unique=True, nullable=False)
    username: Mapped[str | None] = mapped_column(String)
    stars: Mapped[int] = mapped_column(Integer, default=0)

    logs: Mapped[list["ActivityLog"]] = relationship(back_populates="user")


class Exercise(Base):
    __tablename__ = "exercises"

    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(100), unique=True, nullable=False)
    stars_reward: Mapped[int] = mapped_column(default=1)

    logs: Mapped[list["ActivityLog"]] = relationship(back_populates="exercise")


class ActivityLog(Base):
    __tablename__ = "workout_logs"

    id: Mapped[int] = mapped_column(primary_key=True)
    user_id: Mapped[int] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"))
    exercise_id: Mapped[int] = mapped_column(ForeignKey("exercises.id", ondelete="CASCADE"))
    created_at: Mapped[datetime] = mapped_column(server_default=func.now())

    user: Mapped["User"] = relationship(back_populates="logs")
    exercise: Mapped["Exercise"] = relationship(back_populates="logs")

from sqlalchemy import select
from database.base import async_session
from database.models import User, Exercise, ActivityLog


async def set_user(telegram_id: int, username: str | None = None) -> User:
    async with async_session() as session:
        stmt = select(User). where(User.telegram_id == telegram_id)
        result = await session.execute(stmt)
        user = result.scalar_one_or_none()

        if not user:
            user = User(telegram_id=telegram_id, username=username)
            session.add(user)
            await session.commit()
            await session.refresh(user)

        return user


async def get_user(telegram_id: int) -> User:
    async with async_session() as session:
        stmt = select(User).where(User.telegram_id == telegram_id)
        result = await session.execute(stmt)
        return result.scalar_one_or_none()


async def add_stars(telegram_id: int, count: int = 1) -> int:
    async with async_session() as session:
        stmt = select(User).where(User.telegram_id == telegram_id)
        result = await session.execute(stmt)
        user = result.scalar_one_or_none()

        if user:
            user.stars += count
            await session.commit()
            return user.stars
        return 0


async def init_exercises():
    """Заполняет справочник упражнений при первом запуске, если он пуст"""
    initial_data = [
        {"name": "🥗 Порция овощей", "stars_reward": 1},
        {"name": "🍫 0 сахара за день", "stars_reward": 2},
        {"name": "🌭 0 фастфуда и колбасы", "stars_reward": 2},
        {"name": "📊 Норма КБЖУ", "stars_reward": 3},
        {"name": "🍷 0 алкоголя (пт-вс)", "stars_reward": 3},
        {"name": "🏋️ Кроссфит", "stars_reward": 3},
        {"name": "🏃 Бег", "stars_reward": 3},
        {"name": "🏔 Поход (1 день)", "stars_reward": 5},
        {"name": "🧊 Закаливание / холодный душ", "stars_reward": 1},
        {"name": "✨ Свое достижение", "stars_reward": 0},
    ]

    async with async_session() as session:
        #Проверяем есть ли уже записи в таблице Exercise
        stmt = select(Exercise)
        result = await session.execute(stmt)
        existing = result.scalars().all()

        if not existing:
            for item in initial_data:
                exercise = Exercise(name=item["name"], stars_reward=item["stars_reward"])
                session.add(exercise)
            await session.commit()
            print("Справочник упражнений успешно инициализирован")

async def get_all_exercises():
    """Возвращает список всех полезных привычек из базы"""
    async with async_session() as session:
        stmt = select(Exercise)
        result = await session.execute(stmt)
        return result.scalars().all()


async def log_activity(telegram_id: int, exercise_id: int):
    """Записывает выполненную активность в лог и начисляет звезды"""
    async with async_session() as session:
        #Находим юзера
        user_stmt = select(User).where(User.telegram_id == telegram_id)
        user_res = await session.execute(user_stmt)
        user = user_res.scalar_one_or_none()

        #Находим упражнение
        ex_stmt = select(Exercise).where(Exercise.id == exercise_id)
        ex_res = await session.execute(ex_stmt)
        exercise = ex_res.scalar_one_or_none()

        if user and exercise:
            #Создаем запись в логе
            log = ActivityLog(user_id=user.id, exercise_id=exercise_id)
            session.add(log)
            #Начисляем звезды, если это не кастомное достижение в 0 звезд
            if exercise.stars_reward > 0:
                user.stars += exercise.stars_reward

            await session.commit()
            return user.stars, exercise
        return 0, None

# Старт, меню, нажатия на кнопки
from aiogram import Router, F
from aiogram.filters import CommandStart
from aiogram.types import Message, CallbackQuery
from aiogram.fsm.context import FSMContext
from aiogram.fsm.state import State, StatesGroup

from database.requests import get_all_exercises, log_activity, set_user, get_user, add_stars
from keyboards.user_kb import main_kb, get_exercise_kb


user_router = Router()


class CustomAchievement(StatesGroup):
    waiting_for_text = State()


@user_router.message(CommandStart())
async def cmd_start(message: Message):
    #Регистрируем пользователя в базе данных
    user = await set_user(
        telegram_id=message.from_user.id,
        username=message.from_user.username
    )

    await message.answer(
        f"Здорово, {message.from_user.first_name}! 🔥\n\n"
        f"Ты зарегистрирован в системе 'ЗОЖ ачивки'.\n"
        f"Сейчас у тебя звезд: ⭐️ <b>{user.stars}</b>",
        reply_markup=main_kb
    )

@user_router.message(F.text == "⭐️ Мой баланс")
async def show_balance(message: Message):
    user = await get_user(message.from_user.id)
    stars = user.stars if user else 0
    await message.answer(f"твой текущий баланс: ⭐️ <b>{stars}</b> звезд.")


@user_router.message(F.text == "💪 Отметить активность")
async def track_workout(message: Message):
    exercises = await get_all_exercises()
    keyboard = get_exercise_kb(exercises)
    await message.answer("Выбери, что именно ты сделал сегодня:", reply_markup=keyboard)


@user_router.callback_query(F.data.startswith("ex_"))
async def process_Exercise_callback(callback: CallbackQuery, state: FSMContext):
    exercise_id = int(callback.data.split("_")[1])

    #Логируем активность в базу
    new_balance, exercise = await log_activity(
        telegram_id=callback.from_user.id,
        exercise_id=exercise_id
    )

    if exercise:
        if exercise.stars_reward > 0:
            await callback.message.edit_text(
                f"Отлично! Засчитано: <b>{exercise.name}</b> (+{exercise.stars_reward}⭐️)\n"
                f"Твой новый баланс ⭐️ <b>{new_balance}</b>"
            )
        else:
            #Здесь позже сделаем логику для кастомного достижения
            await state.set_state(CustomAchievement.waiting_for_text)

            await callback.message.edit_text(
                "✨ Ты выбрал кастомное достижение!\n"
                "Напиши текстом следующим сообщением, что именно ты сделал."
            )

    await callback.answer()


#Ловим текст, когда бот находится в состоянии ожижания достижения
@user_router.message(CustomAchievement.waiting_for_text)
async def process_custom_achievement_text(message: Message, state: FSMContext):
    achievement_text = message.text

    #Сбрасываем состояние, чтобы бот снова перешел в обычный режим
    await state.clear()

    #Здесь в будущем мы добавим отправку текста администратору
    await message.answer(
        f"✅ Твое достижение принято и отправлено на рассмотрение:\n"
        f"<i>«{achievement_text}»</i>\n\n"
        f"Администратор проверит его и начислит заслуженные звезды!"
    )

from aiogram.types import (KeyboardButton, ReplyKeyboardMarkup, InlineKeyboardMarkup,
                           InlineKeyboardButton, InlineKeyboardMarkup)
from aiogram.utils.keyboard import InlineKeyboardBuilder



#Главное меню пользователя
main_kb = ReplyKeyboardMarkup(
    keyboard=[
        [KeyboardButton(text="💪 Отметить активность"), KeyboardButton(text="⭐️ Мой баланс")],
        [KeyboardButton(text="🏆 Рейтинг")]
    ],
    resize_keyboard=True,
    input_field_placeholder="Выбери действие..."
)


def get_exercise_kb(exercises) -> InlineKeyboardMarkup:
    builder = InlineKeyboardBuilder()
    for ex in exercises:
        builder.row(
            InlineKeyboardButton(
                text=f"{ex.name} +{ex.stars_reward}⭐" if ex.stars_reward > 0 else ex.name, callback_data=f"ex_{ex.id}"
            )
        )
    return builder.as_markup()
